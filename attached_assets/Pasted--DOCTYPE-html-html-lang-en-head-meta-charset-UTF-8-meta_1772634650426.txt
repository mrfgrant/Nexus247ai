<script>
  // Scroll reveal
  const reveals = document.querySelectorAll('.reveal');
  const io = new IntersectionObserver((entries) => {
    entries.forEach((e, i) => {
      if (e.isIntersecting) {
        setTimeout(() => e.target.classList.add('visible'), i * 80);
        io.unobserve(e.target);
      }
    });
  }, { threshold: 0.1 });
  reveals.forEach(el => io.observe(el));

  // Score card animation
  function animateScore() {
    const bars = document.querySelectorAll('.sc-bar-fill');
    const scoreEls = document.querySelectorAll('.sc-row-score');
    const overallEl = document.getElementById('overall-num');

    bars.forEach((bar, i) => {
      const targetWidth = parseInt(bar.dataset.width);
      const targetScore = parseInt(scoreEls[i].dataset.score);
      setTimeout(() => {
        bar.style.width = targetWidth + '%';
        let current = 0;
        const tick = setInterval(() => {
          current++;
          scoreEls[i].textContent = current + '/10';
          if (current >= targetScore) clearInterval(tick);
        }, 90);
      }, 500 + i * 200);
    });

    // overall count-up
    setTimeout(() => {
      let n = 0;
      const target = 86;
      const tick = setInterval(() => {
        n += 2;
        overallEl.innerHTML = Math.min(n, target) + '<span>/100</span>';
        if (n >= target) clearInterval(tick);
      }, 20);
    }, 500 + 5 * 200 + 300);
  }

  window.addEventListener('load', () => setTimeout(animateScore, 700));
</script>
