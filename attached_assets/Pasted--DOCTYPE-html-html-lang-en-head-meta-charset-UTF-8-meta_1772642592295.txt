<script>
  // Disable right-click
  document.addEventListener('contextmenu', e => e.preventDefault());

  // Disable common screenshot keyboard shortcuts (partial deterrent)
  document.addEventListener('keydown', e => {
    if (e.key === 'PrintScreen') { e.preventDefault(); }
    if ((e.ctrlKey || e.metaKey) && e.shiftKey && ['s','S','4','3'].includes(e.key)) { e.preventDefault(); }
    // Disable Ctrl+P (print)
    if ((e.ctrlKey || e.metaKey) && e.key === 'p') { e.preventDefault(); }
    // Disable Ctrl+C on blurred content (user-select:none handles most of this)
  });

  // Modal
  function showUnlock() {
    const m = document.getElementById('unlockModal');
    m.style.display = 'flex';
  }
  function hideUnlock() {
    document.getElementById('unlockModal').style.display = 'none';
  }
  document.getElementById('unlockModal').addEventListener('click', function(e) {
    if (e.target === this) hideUnlock();
  });

  // Animate score ring + dim bars on load
  window.addEventListener('load', () => {
    setTimeout(() => {
      // Ring
      const target = 86;
      const circumference = 251.2;
      const offset = circumference - (target / 100) * circumference;
      document.getElementById('ringFill').style.strokeDashoffset = offset;

      // Count up ring number
      let n = 0;
      const tick = setInterval(() => {
        n += 2;
        document.getElementById('ringNum').textContent = Math.min(n, target);
        if (n >= target) clearInterval(tick);
      }, 18);

      // Dim bars
      const fills = document.querySelectorAll('.dim-fill');
      const scores = document.querySelectorAll('.dim-score');
      fills.forEach((bar, i) => {
        setTimeout(() => {
          bar.style.width = bar.dataset.w + '%';
          let s = 0;
          const st = parseInt(scores[i].dataset.s);
          const si = setInterval(() => {
            s++;
            scores[i].textContent = s;
            if (s >= st) clearInterval(si);
          }, 100);
        }, 300 + i * 150);
      });
    }, 400);
  });
</script>
